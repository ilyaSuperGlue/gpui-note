```ts
const fizzbuzz = (n: number) => {
  if (n % 3 === 0 && n % 5 === 0) {
    console.log("fizzbuzz");
  } else if (n % 3 === 0) {
    console.log("fizz");
  } else if (n % 5 === 0) {
    console.log("buzz");
  } else {
    console.log(`the number ${n} can't be divided by 3 or 5`);
  }
};
Array.from({ length: 100 }, (_, index) => fizzbuzz(index + 1));
```
